## 内存安全的前缀树

### 传统前缀树, 会造成内存泄漏

```cpp
class Trie1 {
private:
    struct TrieNode {
        int pass;
        int end;
        vector<TrieNode*> nexts;

        TrieNode() : pass(0), end(0), nexts(26, nullptr) {}
    };

    TrieNode* root;

public:
    Trie1() {
        root = new TrieNode();
    }

    void insert(string word) {
        TrieNode* node = root;
        node->pass++;
        for (int i = 0; i < word.length(); i++) {
            int path = word[i] - 'a'; // 由字符，对应成走向哪条路
            if (node->nexts[path] == nullptr) {
                node->nexts[path] = new TrieNode();
            }
            node = node->nexts[path];
            node->pass++;
        }
        node->end++;
    }

    // 如果之前word插入过前缀树，那么此时删掉一次
    // 如果之前word没有插入过前缀树，那么什么也不做
    void erase(string word) {
        if (countWordsEqualTo(word) > 0) {
            TrieNode* node = root;
            node->pass--;
            for (int i = 0; i < word.length(); i++) {
                int path = word[i] - 'a';
                if (--node->nexts[path]->pass == 0) {
                    node->nexts[path] = nullptr;
                    return;
                }
                node = node->nexts[path];
            }
            node->end--;
        }
    }

    // 查询前缀树里，word单词出现了几次
    int countWordsEqualTo(string word) {
        TrieNode* node = root;
        for (int i = 0; i < word.length(); i++) {
            int path = word[i] - 'a';
            if (node->nexts[path] == nullptr) {
                return 0;
            }
            node = node->nexts[path];
        }
        return node->end;
    }

    // 查询前缀树里，有多少单词以pre做前缀
    int countWordsStartingWith(string pre) {
        TrieNode* node = root;
        for (int i = 0; i < pre.length(); i++) {
            int path = pre[i] - 'a';
            if (node->nexts[path] == nullptr) {
                return 0;
            }
            node = node->nexts[path];
        }
        return node->pass;
    }
};

```





### 方案一：补全析构函数并修复 `erase`（传统 C++ 做法）

```cpp
class Trie1 {
private:
    struct TrieNode {
        int pass;
        int end;
        vector<TrieNode*> nexts;
        TrieNode() : pass(0), end(0), nexts(26, nullptr) {}
        
        // 添加递归析构函数
        ~TrieNode() {
            for (TrieNode* child : nexts) {
                if (child != nullptr) {
                    delete child; // 递归释放子节点内存
                }
            }
        }
    };

    TrieNode* root;

public:
    Trie1() { root = new TrieNode(); }
    
    // 添加类的析构函数
    ~Trie1() {
        delete root; // 触发 TrieNode 的递归析构
    }

    // ... 其他代码保持不变 ...

    void erase(string word) {
        if (countWordsEqualTo(word) > 0) {
            TrieNode* node = root;
            node->pass--;
            for (int i = 0; i < word.length(); i++) {
                int path = word[i] - 'a';
                if (--node->nexts[path]->pass == 0) {
                    // 修复内存泄漏：delete 会触发递归析构，释放该节点及其所有子节点
                    delete node->nexts[path]; 
                    node->nexts[path] = nullptr;
                    return;
                }
                node = node->nexts[path];
            }
            node->end--;
        }
    }
    // ...
};
```

### 方案二：使用现代 C++ 的智能指针（推荐做法）

```cpp
#include <memory>
#include <vector>
#include <string>

using namespace std;

class Trie1 {
private:
    struct TrieNode {
        int pass;
        int end;
        vector<unique_ptr<TrieNode>> nexts; // 使用 unique_ptr

        TrieNode() : pass(0), end(0), nexts(26) {} // 默认构造为 nullptr
    };

    unique_ptr<TrieNode> root; // 根节点也使用 unique_ptr

public:
    Trie1() : root(new TrieNode()) {}

    void insert(string word) {
        TrieNode* node = root.get();
        node->pass++;
        for (int i = 0; i < word.length(); i++) {
            int path = word[i] - 'a';
            if (node->nexts[path] == nullptr) {
                node->nexts[path] = make_unique<TrieNode>();
            }
            node = node->nexts[path].get();
            node->pass++;
        }
        node->end++;
    }

    void erase(string word) {
        if (countWordsEqualTo(word) > 0) {
            TrieNode* node = root.get();
            node->pass--;
            for (int i = 0; i < word.length(); i++) {
                int path = word[i] - 'a';
                if (--node->nexts[path]->pass == 0) {
                    // 修复内存泄漏：reset 会自动 delete 该节点及其子树
                    node->nexts[path].reset(); 
                    return;
                }
                node = node->nexts[path].get();
            }
            node->end--;
        }
    }

    int countWordsEqualTo(string word) {
        TrieNode* node = root.get();
        for (int i = 0; i < word.length(); i++) {
            int path = word[i] - 'a';
            if (node->nexts[path] == nullptr) {
                return 0;
            }
            node = node->nexts[path].get();
        }
        return node->end;
    }

    int countWordsStartingWith(string pre) {
        TrieNode* node = root.get();
        for (int i = 0; i < pre.length(); i++) {
            int path = pre[i] - 'a';
            if (node->nexts[path] == nullptr) {
                return 0;
            }
            node = node->nexts[path].get();
        }
        return node->pass;
    }
};
```
