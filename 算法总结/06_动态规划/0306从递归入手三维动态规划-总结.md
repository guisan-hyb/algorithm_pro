# 从递归入手三维动态规划

> 来源笔记：`0306从递归入手三维动态规划.cpp`｜核心程度：必背｜分类：06 动态规划

## 一句话

尝试函数有几个可变参数，就对应几维 DP 表；三维只是因为"状态里多带了一个参数"（剩余容量、剩余步数、已用次数、当前余数……），套路仍是**递归 → 记忆化 → 迭代 → 空间压缩**。

## 五道题

### 题一：一和零（LeetCode 474，多维费用背包）

`dp[i][z][o]` = 前 i 个字符串、0 的预算 z、1 的预算 o 时最多能选几个：

```
不要第 i 个：dp[i][z][o] = dp[i+1][z][o]
要第 i 个  ：if (zeros <= z && ones <= o) dp[i+1][z-zeros][o-ones] + 1
```

空间压缩成二维：**z、o 都要倒序遍历**（否则同一物品会被重复使用）。

### 题二：盈利计划（LeetCode 879，多维费用背包 + 计数）

`f(i, n, sum)` = 前 i 个任务、还剩 n 个人、已赚 sum 的方案数；`sum` 一旦超过 `minProfit` 就**截断成 minProfit**（笔记的"优化内存"技巧：因为超过之后只关心"达标"这一件事）。

### 题三：骑士在棋盘上的概率（LeetCode 688）

`dp[step][i][j]` = 走 step 步后仍在 (i,j) 的概率：

- 记忆化：`f(i,j,k)`，越界返回 0，`k == 0` 返回 1，八个方向各 `/8` 累加；
- 迭代：**把棋盘整体平移 2 格**（`2..n+1`），数组开 `n+4`，这样八个方向的跳跃都不会越界；`dp[0][*][*] = 1`，逐层推进。

### 题四：矩阵中和能被 K 整除的路径（LeetCode 2435）

**用同余原理把"路径和"压缩成"余数"**（否则第三维会爆内存）：

- `f(i, j, r)`：在 (i,j)、希望当前位置 `%k` 等于 r 时，到终点有多少条路径；
- 转移：`need = (r - grid[i][j] % k + k) % k`，然后 `f(i+1,j,need) + f(i,j+1,need)`；
- base：终点处 `grid[i][j] % k == r` 返回 1。

### 题五：扰乱字符串（LeetCode 87）

`f(l1, l2, len)` = s1 从 l1 起长 len 的子串 与 s2 从 l2 起长 len 的子串 是否互为扰乱字符串：

- 不交错：枚举分割点 k，`f(l1,l2,k) && f(l1+k,l2+k,len-k)`；
- 交错：`f(l1, l2+len-k, k) && f(l1+k, l2, len-k)`；
- 用 `len` 做第三维，**把四维压成三维**（因为两个子串长度必然相等）；
- 迭代填表按 `len` 从 2 到 n 分层。

## 适用特征（看到这些信号就选它）

- 状态 = 位置 + 额外信息（预算/剩余步数/已用次数/余数/长度）；
- "多维费用背包"（容量不止一个维度）；
- 概率/方案数类，且每步有多种等可能选择。

## 易错点

1. 三维数组先想清楚**哪一维是"层"**（能当层的那一维适合做迭代的外层循环）；
2. 空间压缩时，依赖"上一层同位置 + 上一层左前方" → 两个维度都要**倒序**；
3. 状态压缩技巧（如"超过 minProfit 就截断"）能救命，要先分析"哪些信息其实不再重要"；
4. 用同余代替大数值参数（避免第三维爆炸）；
5. 迭代填表前先做**棋盘/数组平移**，能省掉大量越界判断；
6. 记忆化搜索的 `dp` 哨兵值要和合法答案区分开。

## 模板（默写骨架）

```cpp
#include <bits/stdc++.h>
using namespace std;

// ---------- 1. 一和零（多维费用背包，二维空间压缩） ----------
int findMaxForm(const vector<string>& strs, int m, int n) {
    vector<vector<int>> dp(m + 1, vector<int>(n + 1, 0));
    for (const string& s : strs) {
        int zeros = 0, ones = 0;
        for (char c : s) (c == '0' ? zeros : ones)++;
        for (int z = m; z >= zeros; z--)              // 倒序！
            for (int o = n; o >= ones; o--)
                dp[z][o] = max(dp[z][o], 1 + dp[z - zeros][o - ones]);
    }
    return dp[m][n];
}

// ---------- 2. 盈利计划（记忆化 + 截断优化） ----------
class ProfitableSchemes {
    static const int MOD = 1000000007;
    vector<int> group, profit;
    int minProfit;
    vector<vector<vector<long long>>> dp;
    int f(int i, int n, int sum) {
        if (i < 0) return sum >= minProfit ? 1 : 0;
        if (n <= 0) return sum >= minProfit ? 1 : 0;
        if (sum > minProfit) sum = minProfit;         // 截断，省内存
        if (dp[i][n][sum] != -1) return (int)dp[i][n][sum];
        long long ans = f(i - 1, n, sum);
        if (n >= group[i]) ans += f(i - 1, n - group[i], sum + profit[i]);
        return (int)(dp[i][n][sum] = ans % MOD);
    }
public:
    int profitableSchemes(int n, int mp, vector<int>& g, vector<int>& p) {
        group = g; profit = p; minProfit = mp;
        int m = (int)p.size();
        dp.assign(m, vector<vector<long long>>(n + 1, vector<long long>(mp + 1, -1)));
        return f(m - 1, n, 0);
    }
};

// ---------- 3. 骑士在棋盘上的概率（迭代 + 平移） ----------
double knightProbability(int n, int k, int row, int column) {
    int move[8][2] = {{-1,-2},{-1,2},{-2,-1},{-2,1},{1,-2},{1,2},{2,-1},{2,1}};
    vector<vector<vector<double>>> dp(k + 1,
        vector<vector<double>>(n + 4, vector<double>(n + 4, 0)));
    for (int i = 2; i <= n + 1; i++)
        for (int j = 2; j <= n + 1; j++) dp[0][i][j] = 1;
    for (int step = 1; step <= k; step++)
        for (int i = 2; i <= n + 1; i++)
            for (int j = 2; j <= n + 1; j++) {
                double tmp = 0;
                for (auto& mv : move) tmp += dp[step - 1][i + mv[0]][j + mv[1]] / 8;
                dp[step][i][j] = tmp;
            }
    return dp[k][row + 2][column + 2];
}

// ---------- 4. 矩阵中和能被 K 整除的路径（同余做第三维） ----------
int numberOfPaths(vector<vector<int>>& grid, int k) {
    const int MOD = 1000000007;
    int m = (int)grid.size(), n = (int)grid[0].size();
    vector<vector<vector<int>>> dp(m, vector<vector<int>>(n, vector<int>(k, 0)));
    dp[m - 1][n - 1][grid[m - 1][n - 1] % k] = 1;
    for (int i = m - 2; i >= 0; i--)
        for (int r = 0; r < k; r++)
            dp[i][n - 1][r] = dp[i + 1][n - 1][(r - grid[i][n - 1] % k + k) % k];
    for (int j = n - 2; j >= 0; j--)
        for (int r = 0; r < k; r++)
            dp[m - 1][j][r] = dp[m - 1][j + 1][(r - grid[m - 1][j] % k + k) % k];
    for (int i = m - 2; i >= 0; i--)
        for (int j = n - 2; j >= 0; j--)
            for (int r = 0; r < k; r++) {
                int need = (r - grid[i][j] % k + k) % k;
                dp[i][j][r] = (dp[i + 1][j][need] + dp[i][j + 1][need]) % MOD;
            }
    return dp[0][0][0];
}
```
